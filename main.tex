% __ Algorithm (theme option `algorithms') _____________________________________
\begin{frame}{Scheduler}
	\begin{narrow}[0.72\textwidth]
		\begin{algorithm}[H]
			\caption{Greedy slice assignment}
			\KwIn{Slice demands $d_1, \dots, d_n$; capacity $C$}
			\KwOut{Assignment $a$}
			Sort slices by descending $d_i$\;
			\ForEach{slice $i$}{
				\If{$d_i \le C$}{
					$a_i \gets d_i$\;
					$C \gets C - d_i$\;
				}
			}
			\Return $a$\;
		\end{algorithm}
	\end{narrow}
\end{frame}
